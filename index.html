function onEdit(e) { if (!e) return; var sheet = e.range.getSheet(); if (sheet.getName() === 'Registro_Socios') { var row = e.range.getRow(); var col = e.range.getColumn(); var celda = e.range; // Si Sandra pega coordenadas en Latitud (Columna L = 12) o Longitud (Columna M = 13) if (col === 12 || col === 13) { var val = celda.getDisplayValue().toString().trim(); // Si pegó ambas juntas separadas por coma (ej: "3.6385, -76.1913" o "3,4538, -76,5365") if (val.indexOf(',') !== -1 && col === 12) { var partes = val.split(','); if (partes.length >= 2) { var latVal = limpiarStrCoord(partes[0]); var lonVal = limpiarStrCoord(partes[1]); if (latVal !== null && lonVal !== null) { sheet.getRange(row, 12).setNumberFormat('@').setValue(latVal); sheet.getRange(row, 13).setNumberFormat('@').setValue(lonVal); sheet.getRange(row, 14).setValue("OK (Manual)"); } } } else { // Si pegó celda por celda suelta var normalizada = limpiarStrCoord(val); if (normalizada !== null) { celda.setNumberFormat('@').setValue(normalizada); sheet.getRange(row, 14).setValue("OK (Manual)"); } } } } } // Función auxiliar matemática para limpiar cualquier formato extraño de Google Maps function limpiarStrCoord(str) { if (!str) return null; var s = str.toString().trim(); var esNegativo = s.indexOf('-') !== -1; var soloDigitos = s.replace(/\D/g, ''); if (!soloDigitos || soloDigitos.length < 2) return null; // Reconstruir con el punto decimal en la posición correcta según longitud de enteros var resultado = ""; if (soloDigitos.length > 10) { // Si tiene muchos dígitos por culpa de los puntos de miles de Sheets if (soloDigitos.indexOf('76') === 2 || soloDigitos.indexOf('74') === 2 || soloDigitos.indexOf('75') === 2) { resultado = soloDigitos.slice(0, 2) + "." + soloDigitos.slice(2); // Es longitud ej: 76.500062... } else { resultado = soloDigitos.slice(0, 1) + "." + soloDigitos.slice(1); // Es latitud ej: 3.4793... } } else { // Formato estándar corto if (soloDigitos.length > 2 && (soloDigitos.startsWith('76') || soloDigitos.startsWith('74') || soloDigitos.startsWith('75') || soloDigitos.startsWith('77'))) { resultado = soloDigitos.slice(0, 2) + "." + soloDigitos.slice(2); } else { resultado = soloDigitos.slice(0, 1) + "." + soloDigitos.slice(1); } } var num = parseFloat(resultado); if (isNaN(num)) return null; return esNegativo ? -Math.abs(num) : num; }
